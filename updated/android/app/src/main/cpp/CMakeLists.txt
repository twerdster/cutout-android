cmake_minimum_required(VERSION 3.22)
project(cutout_android LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS cutout_compile=2 cutout_link=1)
set(CMAKE_JOB_POOL_COMPILE cutout_compile)
set(CMAKE_JOB_POOL_LINK cutout_link)
set(CUTOUT_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(CUTOUT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../../../native" "${CMAKE_CURRENT_BINARY_DIR}/cutout-core")
add_library(cutout_android SHARED bridge.cpp)
target_link_libraries(cutout_android PRIVATE cutout_core)
target_compile_features(cutout_android PRIVATE cxx_std_17)
